% ---- níveis de título -------------------------------------------------------
% O MyST escreve as partes do sumário como \section no arquivo principal, e cada
% capítulo, num arquivo incluído, como \section (fora de uma parte) ou como
% \subsection (dentro de uma parte); os títulos internos vêm logo abaixo.
% Aqui o primeiro título de cada arquivo incluído vira \chapter, os seguintes
% descem um nível de cada vez, e as partes viram \part.
\makeatletter
\let\cnLchapter\chapter \let\cnLsection\section \let\cnLsubsection\subsection
\let\cnLsubsubsection\subsubsection \let\cnLparagraph\paragraph
\newif\ifcn@incluido
\newcount\cn@base
\AddToHook{include/before}{\global\cn@incluidotrue\global\cn@base=0\relax}
\AddToHook{include/after}{\global\cn@incluidofalse}
\newcommand{\cn@titulo}[2]{%
  \ifcn@incluido
    \ifnum\cn@base=0 \global\cn@base=#1\relax\fi
    \ifcase\numexpr#1-\cn@base\relax
      \cnLchapter{#2}%
    \or \cnLsection{#2}%
    \or \cnLsubsection{#2}%
    \or \cnLsubsubsection{#2}%
    \else \cnLparagraph{#2}%
    \fi
  \else
    \cleardoublepage\phantomsection\addcontentsline{toc}{part}{#2}\part*{#2}%
  \fi}
\renewcommand{\section}{\@ifstar{\cn@titulo{1}}{\cn@titulo{1}}}
\renewcommand{\subsection}{\@ifstar{\cn@titulo{2}}{\cn@titulo{2}}}
\renewcommand{\subsubsection}{\@ifstar{\cn@titulo{3}}{\cn@titulo{3}}}
\renewcommand{\paragraph}{\@ifstar{\cn@titulo{4}}{\cn@titulo{4}}}
\makeatother
